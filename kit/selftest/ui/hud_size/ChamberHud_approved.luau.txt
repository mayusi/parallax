local function tween(o, t, props, style, dir)
	local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
	tw:Play()
	return tw
end

local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in props do
		o[k] = v
	end
	o.Parent = parent
	return o
end

-- ===== LETTERS: live words from the glyph sheet =====
-- a Frame sized to the word (cap = the cap height in px), one ImageLabel per glyph on one baseline; :set(text) re-sets
-- it, :fade(alpha) sets every glyph's transparency, :tint(colour). align: 0 left (default), 0.5 centre, 1 right of the
-- frame's Position (it is the AnchorPoint's X).
local Letters = {}
Letters.__index = Letters

function ChamberHud.letters(parent, text, cap, colour, align)
	local self = setmetatable({}, Letters)
	self.cap = cap
	self.colour = colour or BONE
	self.alpha = 0
	self.frame = new("Frame", { Name = "Letters", BackgroundTransparency = 1, AnchorPoint = Vector2.new(align or 0, 0) }, parent)
	self:set(text)
	return self
end

function Letters:set(text)
	text = string.upper(tostring(text or ""))
	if text == self.text then return self end
	self.text = text
	for _, k in self.frame:GetChildren() do
		if k:IsA("ImageLabel") then -- the glyphs only: a caller's UIScale (the count's pop) lives here too
			k:Destroy()
		end
	end
	local G = H.Glyphs
	local s = self.cap / G.Cap
	local track = self.cap * 0.1
	local x = 0
	for i = 1, #text do
		local ch = string.sub(text, i, i)
		local g = G.Map[ch]
		if g then
			new("ImageLabel", { Name = "G" .. i, BackgroundTransparency = 1, Image = G.Image,
				ImageRectOffset = Vector2.new(g[1], g[2]), ImageRectSize = Vector2.new(g[3], G.H),
				Position = UDim2.fromOffset(math.floor(x + 0.5), 0), Size = UDim2.fromOffset(math.floor(g[3] * s + 0.5), math.floor(G.H * s + 0.5)),
				ImageColor3 = self.colour, ImageTransparency = self.alpha }, self.frame)
			x += g[3] * s + track
		else
			x += self.cap * 0.42 -- a space (or a mark the sheet does not have)
		end
	end
	self.width = math.max(0, x - track)
	self.frame.Size = UDim2.fromOffset(math.floor(self.width + 0.5), math.floor(G.H * s + 0.5))
	return self
end

function Letters:fade(alpha)
	self.alpha = alpha
	-- (the glyphs only: the frame also holds a UIScale, and setting ImageTransparency on it threw every frame and
	-- aborted the controller's render step while a damage number was up - governor fix 2026-10-27)
	for _, k in self.frame:GetChildren() do
		if k:IsA("ImageLabel") or k:IsA("ImageButton") then
			k.ImageTransparency = alpha
		end
	end
end

function Letters:tint(colour)
	if colour == self.colour then return end
	self.colour = colour
	for _, k in self.frame:GetChildren() do
		if k:IsA("ImageLabel") or k:IsA("ImageButton") then
			k.ImageColor3 = colour
		end
	end
end

-- the cap line's top inside a letters frame of this cap (to line words of different caps up on one baseline)
function ChamberHud.capTop(cap)
	return (H.Glyphs.Base - H.Glyphs.Cap) * cap / H.Glyphs.Cap
end

-- a belt: the rendered strap with its loops (Config.Chamber.Hud.Belt.Plates.<kind>) and a round picture standing in
-- every loop (the round sprite is cut where the loop's front covers it, so it sits IN the loop)
local function buildBelt(parent, name, plate, kind, slotAt)
	local B = H.Belt
	local K = B.K
	local s1 = plate.Slots[1]
	local img = new("ImageLabel", { Name = name, BackgroundTransparency = 1, Image = plate.Image, ZIndex = 2,
		Position = UDim2.fromOffset(slotAt.X - s1[1] * K, slotAt.Y - s1[2] * K), Size = UDim2.fromOffset(plate.Size[1] * K, plate.Size[2] * K) }, parent)
	local r = B.Rounds[kind]
	local slots = {}
	for i, at in plate.Slots do
		local home = UDim2.fromOffset((at[1] - r.Anchor[1]) * K, (at[2] - r.Anchor[2]) * K)
		slots[i] = { home = home, shown = nil, kind = kind,
			img = new("ImageLabel", { Name = "Round" .. i, BackgroundTransparency = 1, Image = r.Image, ZIndex = 3, Position = home,
				Size = UDim2.fromOffset(r.Size[1] * K, r.Size[2] * K) }, img) }
	end
	return { plate = img, slots = slots, right = img.Position.X.Offset + img.Size.X.Offset }
end

-- the first `count` loops hold a round: a round leaving flips up out of its loop (the ammo stacks' spend), one coming
-- back drops in; `kind` swaps the picture (copper / hot / steel), `tint` multiplies it
local function setBelt(belt, count, kind, tint)
	local art = H.Belt.Rounds[kind].Image
	for i, sl in belt.slots do
		local want = i <= count
		local o = sl.img
		if sl.kind ~= kind then
			sl.kind = kind
			o.Image = art
		end
		o.ImageColor3 = tint
		if sl.shown == nil then
			-- the first look: as it is, no motion
			sl.shown = want
			o.ImageTransparency = if want then 0 else 1
		elseif want ~= sl.shown then
			sl.shown = want
			if want then
				o.Rotation = 0
				o.Position = sl.home - UDim2.fromOffset(0, 22)
				o.ImageTransparency = 0
				tween(o, 0.16, { Position = sl.home }, Enum.EasingStyle.Back)
			else
				o.Position = sl.home
				o.Rotation = 0
				tween(o, 0.3, { Position = sl.home + UDim2.fromOffset(14, -46), Rotation = 38, ImageTransparency = 1 })
			end
		end
	end
end

-- a soft dark falloff behind a bottom corner, so the HUD reads on a bright floor: darkest in the screen's corner,
-- fading up and inwards (a CanvasGroup's vertical fade over a frame's horizontal one)
local function shade(parent, pos, size, darkAtLeft)
	local g = new("CanvasGroup", { Name = "Shade", BackgroundTransparency = 1, Position = pos, Size = size, ZIndex = 0 }, parent)
	new("UIGradient", { Rotation = 90, Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 1), NumberSequenceKeypoint.new(0.4, 0.3), NumberSequenceKeypoint.new(1, 0) }) }, g)
	local f = new("Frame", { BackgroundColor3 = Color3.new(0, 0, 0), BorderSizePixel = 0, Size = UDim2.fromScale(1, 1) }, g)
	new("UIGradient", { Rotation = if darkAtLeft then 0 else 180, Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0.2), NumberSequenceKeypoint.new(0.55, 0.6), NumberSequenceKeypoint.new(1, 1) }) }, f)
	return g
end

function ChamberHud:create()
	local playerGui = Players.LocalPlayer:WaitForChild("PlayerGui")
	local gui = new("ScreenGui", { Name = "ChamberHud", ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = 5,

	self.gui = gui
	-- each corner block: its frame, its UIScale and where its ink sits inside the frame (design px from the frame's
	-- edges on the anchored side) - rescale() puts the INK, not the frame, one margin from the screen's edges
	self.corners = {}
	local function corner(name, anchor, size, inset, k)
		local f = new("Frame", { Name = name, BackgroundTransparency = 1, AnchorPoint = anchor, Size = UDim2.fromOffset(size.X, size.Y) }, gui)
		table.insert(self.corners, { frame = f, scale = new("UIScale", { Scale = 1 }, f), anchor = anchor, inset = inset, k = k or 1 })
		return f
	end

	-- VITALS + STYLE, bottom-left (round 8, the owner 10-29: "a bit bigger" - 0.8 of the design; the Spin Drill
	-- moved out to the left edge). The vitality belt at the bottom, the rank above it. With the sword out the GUARD -
	-- Greatsword's blade - slides up out of the vitality into the row on top of it, and the rank rises over it.
	local VW, VH = 1000, 470   -- wide enough for the labels (the guard row is a CanvasGroup: it clips)
	-- until the owner's go on the vertical Spin Drill (DT_VERTICAL) its horizontal bar keeps its place under the
	-- vitality. Ink: the vitality plate starts 141 in; the bottom is its rounds' feet (~17 up) or the bar's (~6)
	self.dtVert = DT_VERTICAL
	local vit = corner("Vitals", Vector2.new(0, 1), Vector2.new(VW, VH), Vector2.new(141, if DT_VERTICAL then 17 else 6), 0.8)
	self.vitals = vit
	self.vitalsCorner = self.corners[#self.corners]
	shade(vit, UDim2.fromOffset(-140, 0), UDim2.fromOffset(VW + 140, VH + 120), true)   -- past the screen edges at any scale
	local B = H.Belt
	local BX = 192                       -- the vitality belt's first loop
	local vitY = if DT_VERTICAL then VH - 42 else VH - 200   -- its strap's middle (over the horizontal bar's row)
	local above = B.Rounds.Copper.Anchor[2] * B.K   -- a round's top above its strap's middle
	local function label(parent, text, belt, y)
		local l = ChamberHud.letters(parent, text, 18, MUTED, 0)
		l.frame.Position = UDim2.fromOffset(belt.right + 12, y - 9 - ChamberHud.capTop(18))
		return l
	end
	self.vitBelt = buildBelt(vit, "Vitality", B.Plates.Vit, "Copper", Vector2.new(BX, vitY))
	label(vit, "VITALITY", self.vitBelt, vitY)
	if not DT_VERTICAL then
		-- the horizontal Spin Drill (round 5/6) under the vitality: the cold bar, the burning one in a clip
		local D = H.Dt
		local KD = 0.29
		local dtW, dtH = D.Size[1] * KD, D.Size[2] * KD
		local dtX, dtY = BX - 26, VH - 50
		self.dtCold = new("ImageLabel", { Name = "SpinDrill", BackgroundTransparency = 1, Image = D.Cold, ZIndex = 2,
			Position = UDim2.fromOffset(dtX, dtY - D.Mid * KD), Size = UDim2.fromOffset(dtW, dtH) }, vit)
		self.dtClip = new("Frame", { Name = "Fill", BackgroundTransparency = 1, ClipsDescendants = true, ZIndex = 3,
			Size = UDim2.fromOffset(0, dtH) }, self.dtCold)
		self.dtHot = new("ImageLabel", { Name = "Hot", BackgroundTransparency = 1, Image = D.Hot, ZIndex = 3,
			Size = UDim2.fromOffset(dtW, dtH) }, self.dtClip)
		self.dtGeom = { w = dtW, x0 = D.Crack[1] * KD, x1 = D.Crack[2] * KD }
		self._dtFill = 0
		label(vit, "SPIN DRILL", { right = dtX + D.Crack[2] * KD + 40 }, dtY)
	end
	-- GUARD: Greatsword flat (the selector's own picture), dark, the lit copy over it in a clip from the hilt, a ghost of
	-- the guard just lost between them (guardHit); Royal Release: the skull hilt's heat over both. Its row sits on top of
	-- the vitality rounds; it slides up out of them (GUARD_RISE) as the sword comes out.
	local Rb = H.Greatsword
	local KG = 0.62
	local gw, gh = Rb.Size[1] * KG, Rb.Size[2] * KG
	local guardY = vitY - above - 12 - Rb.Ink[4] * KG / 2   -- the blade's middle: its ink 12 over the rounds' tops
	self.guardRise = vitY - guardY
	self.guardRow = new("CanvasGroup", { Name = "Guard", BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1), GroupTransparency = 1, Visible = false }, vit)
	self.guardBase = UDim2.fromOffset(BX - 8 - Rb.Ink[1] * KG, guardY - (Rb.Ink[2] + Rb.Ink[4] / 2) * KG)
	self.guardDark = new("ImageLabel", { Name = "Greatsword", BackgroundTransparency = 1, Image = Rb.Image, ZIndex = 2,
		ImageColor3 = Color3.fromRGB(56, 56, 62), Position = self.guardBase, Size = UDim2.fromOffset(gw, gh) }, self.guardRow)
	self.guardGhostClip = new("Frame", { Name = "Ghost", BackgroundTransparency = 1, ClipsDescendants = true, ZIndex = 3,
		Size = UDim2.fromOffset(0, gh) }, self.guardDark)
	self.guardGhost = new("ImageLabel", { Name = "Ghost", BackgroundTransparency = 1, Image = Rb.Image, ZIndex = 3,
		ImageColor3 = Color3.fromRGB(255, 170, 90), ImageTransparency = 0.2, Size = UDim2.fromOffset(gw, gh) }, self.guardGhostClip)
	self.guardClip = new("Frame", { Name = "Lit", BackgroundTransparency = 1, ClipsDescendants = true, ZIndex = 4,
		Size = UDim2.fromOffset(gw, gh) }, self.guardDark)
	self.guardLit = new("ImageLabel", { Name = "Steel", BackgroundTransparency = 1, Image = Rb.Image, ZIndex = 4,
		Size = UDim2.fromOffset(gw, gh) }, self.guardClip)
	self.guardHeat = new("ImageLabel", { Name = "Heat", BackgroundTransparency = 1, Image = H.Guard.Heat, ZIndex = 5,
		ImageTransparency = 1, Size = UDim2.fromOffset(gw, gh) }, self.guardDark)
	self.guardGeom = { b0 = H.Guard.Blade[1] * KG, b1 = H.Guard.Blade[2] * KG }
	self._guardFill = self.guardGeom.b1
	self._ghostFill, self._ghostHold, self._hitT, self._hitK = 0, 0, 0, 0
	label(self.guardRow, "GUARD", { right = self.guardBase.X.Offset + (Rb.Ink[1] + Rb.Ink[3]) * KG }, guardY)
	-- the rank: its letter over the vitality belt's left end, its name beside it, both above the rounds; they rise over
	-- the guard with the sword
	local upper = new("Frame", { Name = "Upper", BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1) }, vit)
	self.upper = upper
	self.styleFrame = vit
	self.styleFade = new("CanvasGroup", { Name = "Style", BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1), GroupTransparency = 1, ZIndex = 4 }, upper)
	self.rankAt = Vector2.new(224, vitY - above - 78)
	self.nameAt = Vector2.new(BX - B.Plates.Vit.Slots[1][1] * B.K + 236, vitY - above - 8)
	-- the letters: one big, and two small ones for SS / SSS stepping down-right off it (the character-action ligature, in brass)
	self.letterMain = new("ImageLabel", { Name = "Rank", BackgroundTransparency = 1, AnchorPoint = Vector2.new(0.5, 0.5), ZIndex = 5,
		Position = UDim2.fromOffset(self.rankAt.X, self.rankAt.Y), Size = UDim2.fromOffset(170, 170) }, self.styleFade)
	self.letterScale = new("UIScale", { Scale = 1 }, self.letterMain)
	self.letterSmall = {}
	for i = 1, 2 do
		self.letterSmall[i] = new("ImageLabel", { Name = "Rank" .. (i + 1), BackgroundTransparency = 1, AnchorPoint = Vector2.new(0.5, 0.5), ZIndex = 6 - i,
			Image = H.Ranks.S.Image, Visible = false, Size = UDim2.fromOffset(86, 86) }, self.styleFade)
	end
	self.rankName = new("ImageLabel", { Name = "RankName", BackgroundTransparency = 1, ZIndex = 5, Size = UDim2.fromOffset(400, 40) }, self.styleFade)
	self._shownRank = nil
	-- the top of the whole bottom-left with the sword out (the rank letter's), for the Spin Drill to stay clear of
	self.vitTop = { y = self.rankAt.Y - 70 - self.guardRise, bottom = VH - 17 }

	-- SPIN DRILL (round 8, the owner: "make it vertical and put it on the left side of the screen"): the obsidian bar
	-- standing under its horned emblem at the left edge, 38-62% of the height; the crack fills bottom -> top as the
	-- gauge charges, the heart-stone and the horn tips light once there is any; on, the whole crack burns and breathes.
	-- Sized in rescale() (by the screen's height, not a UIScale).
	if DT_VERTICAL then
		local V = H.DtV
		self.dtFrame = new("ImageLabel", { Name = "SpinDrill", BackgroundTransparency = 1, Image = V.Cold, ZIndex = 2 }, gui)
		-- round 9 (3mf9): on, the crack burns down as the gauge drains; what it has spent stays a dim cooling glow
		self.dtSpentClip = new("Frame", { Name = "Spent", BackgroundTransparency = 1, ClipsDescendants = true, ZIndex = 3 }, self.dtFrame)
		self.dtSpent = new("ImageLabel", { Name = "Hot", BackgroundTransparency = 1, Image = V.Hot, ZIndex = 3, ImageTransparency = 1 }, self.dtSpentClip)
		self.dtClip = new("Frame", { Name = "Fill", BackgroundTransparency = 1, ClipsDescendants = true, ZIndex = 3 }, self.dtFrame)
		self.dtHot = new("ImageLabel", { Name = "Hot", BackgroundTransparency = 1, Image = V.Hot, ZIndex = 3 }, self.dtClip)
		self.dtHead = new("ImageLabel", { Name = "Head", BackgroundTransparency = 1, Image = V.Head, ZIndex = 4, ImageTransparency = 1,
			Size = UDim2.fromScale(1, 1) }, self.dtFrame)
		-- in fractions of the picture's height: the crack's top and foot
		self.dtGeom = { c0 = V.Crack[1] / V.Size[2], c1 = V.Crack[2] / V.Size[2], aspect = V.Size[1] / V.Size[2], pad = 28 / V.Size[1] }
		self._dtFill = 0

	-- THE TRIGGER'S EDGE (round 9, 3mf9): the screen's edges flush the Spin Drill's red - a hard pulse on the trigger,
	-- then a low heartbeat while it lasts. Its own ScreenGui under the HUD, edge to edge (no safe-area inset). Settings'

	self.edges = {}
	for _, e in {
		{ UDim2.fromScale(1, 0.24), UDim2.fromScale(0, 0), 90 },    -- top: the red at the edge, gone 24% in

		table.insert(self.edges, f)
	end
	self._edge, self._edgeFlash, self._wasActive = 0, 0, false
	self._dtTone, self._toneShown, self._letterBase = 0, 0, Color3.new(1, 1, 1)

	-- SETTINGS in game (R34): P / Start opens the menu's settings screen over the fight (once per session)
	local okPanel, Panel = pcall(function()
		return require(game:GetService("ReplicatedStorage"):WaitForChild("MainMenu"):WaitForChild("Assets"):WaitForChild("SettingsPanel"))

	self.armsW = W
	shade(arms, UDim2.fromOffset(-120, 0), UDim2.fromOffset(W + 260, 520), false)
	self.armsPic = {
		guns = new("ImageLabel", { Name = "EbonyIvory", BackgroundTransparency = 1, Image = H.Crest.Image, ZIndex = 2 }, arms),
		sword = new("ImageLabel", { Name = "Greatsword", BackgroundTransparency = 1, Image = H.Greatsword.Image, ZIndex = 2 }, arms),

	self:_arms(false, true)
	local ammo = new("Frame", { Name = "Body", BackgroundTransparency = 1, Position = UDim2.fromOffset(0, 150), Size = UDim2.fromOffset(W, 250) }, arms)
	-- each gun's six rounds, lying in a stack: Onyx's left (pointing left), Pearl's right

			Position = UDim2.fromOffset(x, 46), Size = UDim2.fromOffset(96, 180) }, ammo)

			-- the top round fires first: round 1 is the top of the stack

			local r = new("ImageLabel", { Name = "Round" .. i, BackgroundTransparency = 1, Image = C.Round, AnchorPoint = Vector2.new(0.5, 0.5),
				Position = home, Size = UDim2.fromOffset(84, 34), Rotation = if hand == "L" then 180 else 0, ZIndex = 2 + i }, col)
			rounds[i] = { img = r, home = home, rot = r.Rotation }

		local tag = ChamberHud.letters(col, if hand == "L" then "ONYX" else "PEARL", 17, MUTED, 0.5)
		tag.frame.Position = UDim2.fromOffset(48, 14 + per * 22 - 2)
		self.mags[hand] = { col = col, rounds = rounds, shown = per }
	end
	-- the count (brass), the reserve, the status, the gunslinger chip: a column right-aligned left of the stacks
	local cx = W - 228
	self.status = ChamberHud.letters(ammo, "", 20, CRIMSON_TINT, 1)
	self.status.frame.Position = UDim2.fromOffset(cx, 46)
	self.count = ChamberHud.letters(ammo, tostring(self.magSize), 64, BRASS_TINT, 1)
	self.count.frame.Position = UDim2.fromOffset(cx, 76)
	self.countScale = new("UIScale", { Scale = 1 }, self.count.frame)
	self.reserve = ChamberHud.letters(ammo, "", 18, C.Gold, 1)
	self.reserve.frame.Position = UDim2.fromOffset(cx, 164)
	self.gsChip = ChamberHud.letters(ammo, "", 18, C.Gold, 1)
	self.gsChip.frame.Position = UDim2.fromOffset(cx, 196)
	-- the sword's lines: in the stacks' place
	self.swordLine = ChamberHud.letters(ammo, "BLOCK  -  RMB", 20, BONE, 1)
	self.swordLine.frame.Position = UDim2.fromOffset(W, 60)
	self.swordLine.frame.Visible = false
	self.releaseLine = ChamberHud.letters(ammo, "", 20, C.Gold, 1)
	self.releaseLine.frame.Position = UDim2.fromOffset(W, 92)
	self.releaseLine.frame.Visible = false

	-- SCORE + BANNER (top-centre)
	local top = corner("Match", Vector2.new(0.5, 0), Vector2.new(1100, 300), Vector2.new(0, 0))
	self.scoreRound = ChamberHud.letters(top, "", 20, C.Gold, 0.5)
	self.scoreRound.frame.Position = UDim2.fromOffset(550, 2)
	self.scoreLine = new("TextLabel", { Name = "Score", AnchorPoint = Vector2.new(0.5, 0), Position = UDim2.fromOffset(550, 32), Size = UDim2.fromOffset(1000, 26),
		BackgroundTransparency = 1, Text = "", Font = Enum.Font.JosefinSans, TextSize = 22, TextColor3 = BONE, RichText = true,
		TextStrokeTransparency = 0.5 }, top)
	self.bannerGroup = new("CanvasGroup", { Name = "Banner", BackgroundTransparency = 1, AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset(550, 170), Size = UDim2.fromOffset(1100, 110), GroupTransparency = 1 }, top)

	new("ImageLabel", { Name = "BandL", BackgroundTransparency = 1, Image = H.FeedPlate, Position = UDim2.fromOffset(0, 19), Size = UDim2.fromOffset(551, 72) }, self.bannerGroup)
	new("ImageLabel", { Name = "BandR", BackgroundTransparency = 1, Image = H.FeedPlate, Position = UDim2.fromOffset(549, 19), Size = UDim2.fromOffset(551, 72),
		Rotation = 180 }, self.bannerGroup)
	self.banner = ChamberHud.letters(self.bannerGroup, "", 44, BONE, 0.5)
	self.banner.frame.Position = UDim2.fromOffset(550, 55 - 22 - ChamberHud.capTop(44))
	local ms = ReplicatedStorage:FindFirstChild("MatchState")
	if ms then
		self._matchConn = ms.OnClientEvent:Connect(function(p)
			if type(p) == "table" then
				self:showMatch(p)
			end
		end)
	end

	-- the whole HUD follows the safe area's height (the 1080p design: the owner 10-29 wants it "like they were before",
	-- not R2's 0.58); each corner's own k on top (vitality + guard 0.8). Never under 0.42 so a phone can read it, never
	-- so wide that the two bottom blocks (~800 + ~560 design px) meet. Every block's ink sits one margin (30 px at
	-- 1080p, 12 min) from the safe area's edges. The Spin Drill is sized by the height: 38-62% of it, lifted clear
	-- of the bottom-left on short screens (it starts higher there, keeping its 24%).
	local MARGIN = 30
	local function rescale()
		local sz = gui.AbsoluteSize
		if sz.X < 1 or sz.Y < 1 then return end
		local m = math.max(12, math.floor(MARGIN * sz.Y / 1080 + 0.5))
		local s = math.max(0.42, math.min(math.clamp(sz.Y / 1080, 0.5, 1.5), (sz.X - 3 * m) / 1450))
		-- Settings (R34) Access.HudScale, 0.75-1.25; the width cap still wins so the corners never meet
		local user = math.clamp(tonumber(Players.LocalPlayer:GetAttribute("S_Access_HudScale")) or 1, 0.75, 1.25)
		s = math.min(s * user, math.max(0.42, (sz.X - 3 * m) / 1450))
		for _, c in self.corners do
			local cs = s * c.k
			c.scale.Scale = cs
			local ax, ay = c.anchor.X, c.anchor.Y
			local x = if ax == 0 then m - c.inset.X * cs elseif ax == 1 then -(m - c.inset.X * cs) else 0
			local y = if ay == 1 then -(m - c.inset.Y * cs) else m * 0.5
			c.frame.Position = UDim2.new(ax, x, ay, y)
		end
		self.hudScale, self.hudMargin = s, m
		if not self.dtVert then return end
		-- the Spin Drill: 38-62% of the height, its foot kept one margin over the bottom-left's top (sword out)
		local vs = s * self.vitalsCorner.k
		local vitTop = sz.Y - m - (self.vitTop.bottom - self.vitTop.y) * vs
		local foot = math.min(sz.Y * 0.62, vitTop - m)
		local top = math.max(2 * m, math.min(sz.Y * 0.38, foot - sz.Y * 0.24))   -- short screens: it starts higher
		local h = math.max(0, foot - top)
		local g = self.dtGeom
		local w = h * g.aspect
		self.dtFrame.Size = UDim2.fromOffset(w, h)
		self.dtFrame.Position = UDim2.fromOffset(m - g.pad * w, top)
		self.dtHot.Size = UDim2.fromOffset(w, h)
		self.dtSpent.Size = UDim2.fromOffset(w, h)
		self.dtH = h
		self.dtFrame.Visible = h >= 24
	end
	self._rescale = rescale
	rescale()
	self._vpConn = gui:GetPropertyChangedSignal("AbsoluteSize"):Connect(rescale)
	self._scaleConn = Players.LocalPlayer:GetAttributeChangedSignal("S_Access_HudScale"):Connect(rescale)
end

-- the MATCH payload: the banner stamps in on its band, the score line under the round
function ChamberHud:showMatch(p)
	if not self.gui then return end
	local msg = tostring(p.message or "")
	if msg ~= "" then
		self.banner:set(msg)
		self.bannerGroup.GroupTransparency = 0
		self.bannerScale.Scale = 1.45
		tween(self.bannerScale, 0.22, { Scale = 1 }, Enum.EasingStyle.Back)
		self._bannerToken = (self._bannerToken or 0) + 1
		local token = self._bannerToken
		task.delay(2.2, function()
			if self.gui and self._bannerToken == token then
				tween(self.bannerGroup, 0.5, { GroupTransparency = 1 })
			end
		end)
	end
	if type(p.rounds) == "table" then
		local parts = {}
		for name, r in p.rounds do
			table.insert(parts, string.format('%s  <font color="#d6a848">%d</font> <font color="#96969e">(%d)</font>',
				string.upper(tostring(name)), r, (p.kills or {})[name] or 0))
		end
		table.sort(parts)
		self.scoreRound:set("ROUND " .. tostring(p.round or 1))
		self.scoreLine.Text = table.concat(parts, "      ")
	else
		self.scoreRound:set("")
		self.scoreLine.Text = ""
	end
end

-- one round leaves its stack: it flips up and away like the menu's ejected brass
function ChamberHud:_spend(hand, i)
	local r = self.mags[hand].rounds[i]
	if not r then return end
	r.img.Position = r.home
	r.img.Rotation = r.rot
	local dir = if hand == "L" then -1 else 1
	tween(r.img, 0.3, { Position = r.home + UDim2.fromOffset(dir * 26, -40), Rotation = r.rot + dir * 70, ImageTransparency = 1 })
end

function ChamberHud:_refill(hand, i)
	local r = self.mags[hand].rounds[i]
	if not r then return end
	r.img.Rotation = r.rot
	r.img.Position = r.home + UDim2.fromOffset(0, 16)
	r.img.ImageTransparency = 0
	tween(r.img, 0.14, { Position = r.home }, Enum.EasingStyle.Back)
end

-- show n rounds in a stack: the top ones are the spent ones
function ChamberHud:_showStack(hand, n)
	local m = self.mags[hand]
	local per = #m.rounds
	n = math.clamp(n, 0, per)
	if n == m.shown then return end
	for i = 1, per do
		local had = i > per - m.shown
		local has = i > per - n
		if had and not has then
			self:_spend(hand, i)
		elseif has and not had then
			self:_refill(hand, i)
		end
	end
	m.shown = n
end

-- the selector: a picture fitted by its ink into a box, the ink's right edge at `right`, its middle at `cy`
local function fit(spec, boxW, boxH)
	local ink = spec.Ink
	local k = math.min(boxW / ink[3], boxH / ink[4])
	return k, ink[3] * k
end

local function placed(spec, k, right, cy)
	local ink = spec.Ink
	return UDim2.fromOffset(spec.Size[1] * k, spec.Size[2] * k),
		UDim2.fromOffset(right - (ink[1] + ink[3]) * k, cy - (ink[2] + ink[4] / 2) * k)
end

function ChamberHud:_arms(sword, snap)
	local W = self.armsW
	local on, off = if sword then "sword" else "guns", if sword then "guns" else "sword"
	local spec = { guns = H.Crest, sword = H.Greatsword }
	local k1, w1 = fit(spec[on], 330, 112)
	local k2, w2 = fit(spec[off], 170, 54)
	local offRight = W - w1 - 30
	local s1, p1 = placed(spec[on], k1, W, 58)
	local s2, p2 = placed(spec[off], k2, offRight, 66)
	local big, small = self.armsPic[on], self.armsPic[off]
	big.ZIndex, small.ZIndex = 3, 2
	local litProps = { Size = s1, Position = p1, ImageColor3 = Color3.new(1, 1, 1), ImageTransparency = 0 }
	local dimProps = { Size = s2, Position = p2, ImageColor3 = Color3.fromRGB(140, 140, 148), ImageTransparency = 0.15 }
	if snap then
		for prop, v in litProps do big[prop] = v end
		for prop, v in dimProps do small[prop] = v end
	else
		tween(big, 0.2, litProps, Enum.EasingStyle.Back)
		tween(small, 0.16, dimProps)
	end
	self.armsKey.frame.Position = UDim2.fromOffset(offRight - w2 - 14, 66 - 12 - ChamberHud.capTop(24))
end

			if not self._sword then g.Visible = false end

function ChamberHud:guardBreak()
	self._breakT = 0.6
end

-- a blocked shot (the governor's sword code, card 5k2f): `drain` = the guard points it took (0-100). The blade flashes,
-- shakes 2-3 px (screen) and leaves a ghost of the guard lost that drains away after a beat; bigger drains hit harder.
function ChamberHud:guardHit(drain)
	if not self.guardRow or not self.guardRow.Visible then return end   -- no guard on screen: nothing to feel
	drain = math.clamp(tonumber(drain) or 0, 0, 100)
	local k = math.clamp(drain / 25, 0.35, 1)
	self._hitT, self._hitK = 0.18 + 0.1 * k, k
	self._ghostHold = math.max(self._ghostHold, 0.22)
end

function ChamberHud:_vitals(ctl, dt)
	local hum = ctl.character and ctl.character:FindFirstChildOfClass("Humanoid")
	local hp = if hum and hum.MaxHealth > 0 then math.clamp(hum.Health / hum.MaxHealth, 0, 1) else 1
	self._hp = hp
	local k = 0.5 + 0.5 * math.sin(self._pulse)
	local WHITE = Color3.new(1, 1, 1)
	-- VITALITY: a round for every twelfth (a chip keeps the round until a whole twelfth is gone)
	local n = #self.vitBelt.slots
	local low = hp < 0.3 and hp > 0
	setBelt(self.vitBelt, math.ceil(hp * n - 1e-6), "Copper", if low then warnPulse(k) else WHITE)

	-- the Spin Drill (StyleService's DTGauge / DTActive on the player): the crack burns up from its foot as far as
	-- the gauge, the heart and the horns' tips as soon as there is any; on, the whole crack burns and breathes
	local plr = Players.LocalPlayer
	local gauge = math.clamp((plr:GetAttribute("DTGauge") or 0) / 100, 0, 1)
	local active = self._dtOn == true
	local ready = gauge >= 0.5
	local dg = self.dtGeom
	if self.dtVert then
		-- the lit height, as a fraction of the picture's from its bottom: the gauge's, on or off - on, it is the DRAIN (the
		-- crack burns down toward its foot as the form uses it up), with the spent length above it a dim cooling glow
		local want = if gauge <= 0 then 0 else (1 - dg.c1) + (dg.c1 - dg.c0) * gauge
		if active and gauge >= 0.999 then want = 1 end   -- the instant of the trigger: the whole bar, the heart included
		self._dtFill += (want - self._dtFill) * math.min(1, dt * 10)
		local h = self.dtH or 0
		local w = self.dtFrame.Size.X.Offset
		local fh = self._dtFill * h
		self.dtClip.Size = UDim2.fromOffset(w, fh)
		self.dtClip.Position = UDim2.fromOffset(0, h - fh)
		self.dtHot.Position = UDim2.fromOffset(0, fh - h)
		local sh = (1 - dg.c0) * h                       -- the spent glow: the whole crack, under the live fill
		self.dtSpentClip.Size = UDim2.fromOffset(w, sh)
		self.dtSpentClip.Position = UDim2.fromOffset(0, h - sh)
		self.dtSpent.Position = UDim2.fromOffset(0, sh - h)
		self._spentGlow = (self._spentGlow or 0) + ((if active then 1 else 0) - (self._spentGlow or 0)) * math.min(1, dt * 4)
		self.dtSpent.ImageTransparency = 1 - self._spentGlow * (0.22 + 0.08 * k)
		self.dtHead.ImageTransparency = if active then 0.25 * (1 - k) elseif gauge > 0 then 0.05 else 1
	else
		local want = if active then dg.w elseif gauge <= 0 then 0 else dg.x0 + (dg.x1 - dg.x0) * gauge
		self._dtFill += (want - self._dtFill) * math.min(1, dt * 10)
		self.dtClip.Size = UDim2.fromOffset(self._dtFill, self.dtClip.Size.Y.Offset)
	end
	self.dtHot.ImageTransparency = if active then 0.3 * (1 - k) elseif ready then 0.18 * (1 - k) else 0

	-- the edge: a pulse on the trigger (decays in ~0.7 s), then a heartbeat (~1.1 s) while the form lasts
	local calm = plr:GetAttribute("S_Access_ReduceFlashing") == true
	if active and not self._wasActive and not calm then
		self._edgeFlash = 1
	end
	self._wasActive = active
	self._edgeFlash *= math.exp(-dt * 4.5)
	self._edgeBeat = ((self._edgeBeat or 0) + dt / 1.1) % 1
	local beat = if calm then 0 else math.exp(-self._edgeBeat * 7) * 0.35   -- a sharp rise, a slow fall
	self._edge += ((if active then 0.55 + 0.45 * beat else 0) - self._edge) * math.min(1, dt * (if calm then 2.5 else 8))
	local e = math.clamp(math.max(self._edge, self._edgeFlash), 0, 1)
	self.edgeGui.Enabled = e > 0.01 and self.gui.Enabled
	if self.edgeGui.Enabled then
		for _, f in self.edges do
			f.BackgroundTransparency = 1 - 0.85 * e
		end
	end

	-- the GUARD + RELEASE (Greatsword.lua's Guard / GuardBroken / Release): the blade's steel lit hilt-to-tip by the
	-- guard left, crimson on a break; the skull hilt heats as Release stores
	if self._breakT > 0 then
		self._breakT -= dt
	end
	if self.guardRow.Visible then
		local g = math.clamp((plr:GetAttribute("Guard") or 100) / 100, 0, 1)
		local broken = plr:GetAttribute("GuardBroken") == true or self._breakT > 0
		local flash = math.sin(os.clock() * 40) > 0
		local gg = self.guardGeom
		local gw = if broken then gg.b0 else gg.b0 + (gg.b1 - gg.b0) * g
		self._guardFill += (gw - self._guardFill) * math.min(1, dt * 12)
		self.guardClip.Size = UDim2.fromOffset(self._guardFill, self.guardClip.Size.Y.Offset)
		-- a blocked shot (guardHit): the ghost holds the guard just lost, then drains to the fill; the blade flashes
		-- and shakes, 2-3 screen px
		-- the ghost: whenever the guard's own value drops (a hit, a break), it keeps what was there - from the raw
		-- value, so it never starts low however the attribute and the hit call are ordered
		-- only a real loss arms it: a shot (20 points) or the break - a held guard's steady 1-point drain must not, or the
		-- ghost sat at full width the whole hold (Play, 10-31: ghost 622 px over a 573 px fill)
		local raw = plr:GetAttribute("Guard") or 100
		local struck = (self._guardRaw or raw) - raw >= GHOST_MIN_DROP or (broken and not self._wasBroken)
		if struck and gw < (self._guardWant or gw) - 0.5 then
			self._ghostFill = math.max(self._ghostFill, self._guardWant)
			self._ghostHold = math.max(self._ghostHold, 0.22)
		end
		self._guardRaw, self._wasBroken = raw, broken
		self._guardWant = gw
		if self._ghostHold > 0 then
			self._ghostHold -= dt
		else
			self._ghostFill += (self._guardFill - self._ghostFill) * math.min(1, dt * 7)
		end
		self.guardGhostClip.Size = UDim2.fromOffset(math.max(self._ghostFill, self._guardFill), self.guardGhostClip.Size.Y.Offset)
		local hitting = self._hitT > 0
		if hitting then
			self._hitT -= dt
			local px = (2 + self._hitK) / math.max(0.1, self.vitalsCorner.scale.Scale)   -- screen px -> design px
			local a = math.max(0, self._hitT) / (0.18 + 0.1 * self._hitK)
			self.guardDark.Position = self.guardBase + UDim2.fromOffset((math.random() * 2 - 1) * px * a, (math.random() * 2 - 1) * px * a)
		else
			self.guardDark.Position = self.guardBase
		end
		self.guardDark.ImageColor3 = if broken and flash then warnTint():Lerp(Color3.new(0, 0, 0), 0.42) else Color3.fromRGB(56, 56, 62)
		-- the guard just lost flashes bright for the hit's first beat, then reads amber while it drains (the unlit
		-- blade stays dark: a pale flash there read as a FULL guard - guardhit_strip.py, 10-29)
		local lit = hitting and self._hitT > 0.1
		self.guardGhost.ImageColor3 = if lit then Color3.fromRGB(255, 245, 220) else Color3.fromRGB(255, 170, 90)
		self.guardGhost.ImageTransparency = if lit then 0 else 0.2
		self.guardLit.ImageColor3 = if g < 0.25 and not broken then warnPulse(k) else WHITE
		local rel = math.clamp(plr:GetAttribute("Release") or 0, 0, 100)
		local hot = rel >= RELEASE_MIN
		self.guardHeat.ImageTransparency = if rel <= 0 then 1 elseif hot then 0.05 + 0.25 * (1 - k) else 1 - rel / RELEASE_MIN * 0.5
		self.releaseLine:set(if broken then "GUARD BROKEN" elseif hot then "ROYAL RELEASE  -  RMB" else "")
		self.releaseLine:tint(if broken then warnTint() else C.Gold)
	end
end

function ChamberHud:_style()
	local plr = Players.LocalPlayer
	local rank = plr:GetAttribute("StyleRank")
	local pts = plr:GetAttribute("StylePoints") or 0
	local show = rank ~= nil and pts > 0
	if show ~= self._styleShown then
		self._styleShown = show
		tween(self.styleFade, if show then 0.15 else 0.6, { GroupTransparency = if show then 0 else 1 })
	end
	if not show then
		self._shownRank = nil
		self._rankIndex = 0
		return
	end
	local i = RANK_AT[rank] or 1
	if rank == self._shownRank then return end
	local up = (self._rankIndex or 0) < i
	self._shownRank = rank
	self._rankIndex = i
	-- the letter: centred by its ink at rankAt; SS / SSS shift it left and add small S's down-right of it
	local art = H.Ranks[string.sub(rank, 1, 1)] or H.Ranks.D
	local multi = #rank > 1
	local lk = 0.86   -- the letters were drawn for a 250 px cylinder; they stand alone now, a little smaller
	local box = (if multi then 176 else 196) * lk
	local ink = art.Ink
	local cx, cy = (ink[1] + ink[3] / 2) / 256, (ink[2] + ink[4] / 2) / 256
	local main = UDim2.fromOffset(self.rankAt.X, self.rankAt.Y) + (if multi then UDim2.fromOffset(-30 * lk, -10 * lk) else UDim2.fromOffset(0, 0))
	self.letterMain.Image = art.Image
	self.letterMain.Size = UDim2.fromOffset(box, box)
	self.letterMain.Position = main + UDim2.fromOffset((0.5 - cx) * box, (0.5 - cy) * box)
	for s = 1, 2 do
		local l = self.letterSmall[s]
		l.Visible = #rank > s
		l.Size = UDim2.fromOffset(100 * lk, 100 * lk)
		l.Position = main + UDim2.fromOffset((70 + (s - 1) * 46) * lk, (34 + (s - 1) * 36) * lk)
	end
	-- the name: its ink's cap 26 px, its ink's left edge and bottom at nameAt (over the vitality rounds)
	local nm = H.Names[rank] or H.Names.D
	local nk = 26 / nm.Ink[4]
	self.rankName.Image = nm.Image
	self.rankName.Size = UDim2.fromOffset(1024 * nk, 96 * nk)
	self.rankName.Position = UDim2.fromOffset(self.nameAt.X - nm.Ink[1] * nk, self.nameAt.Y - (nm.Ink[2] + nm.Ink[4]) * nk)
	local high = i >= 5
	self._letterBase = if high then Color3.fromRGB(255, 200, 170) else Color3.new(1, 1, 1)
	self.letterMain.ImageColor3 = self._letterBase
	self._toneShown = -1   -- (the Spin Drill's tone goes back on over the new letter)
	if up then
		-- a rank up: the letter stamps in
		self.letterScale.Scale = 1.6
		tween(self.letterScale, 0.2, { Scale = 1 }, Enum.EasingStyle.Back)
	else
		self.letterScale.Scale = 0.85
		tween(self.letterScale, 0.25, { Scale = 1 })
	end
end

function ChamberHud:_ammo(ctl)
	local ws = ctl.weaponSystem
	if not ws or self._sword then return end
	local ammo = math.clamp(ws.currentAmmo or 0, 0, self.magSize)
	local ft = ws.handFireTime or {}
	if ammo ~= self._shownAmmo then
		local per = #self.mags.L.rounds
		if self._shownAmmo < 0 or ammo > self._shownAmmo then
			-- a reload (or the first look): both stacks even, Onyx's first
			self._split.L = math.min(per, math.ceil(ammo / 2))
			self._split.R = math.min(per, ammo - self._split.L)
		else
			-- the shots: from the stack of the hand whose timer moved (handFireTime), else the fuller one
			for _ = 1, self._shownAmmo - ammo do
				local h
				if (ft.L or 0) > self._hand.L and self._split.L > 0 then
					h = "L"
					self._hand.L = ft.L
				elseif (ft.R or 0) > self._hand.R and self._split.R > 0 then
					h = "R"
					self._hand.R = ft.R
				else
					h = if self._split.L >= self._split.R then "L" else "R"
				end
				self._split[h] = math.max(0, self._split[h] - 1)
			end
			self.countScale.Scale = 1.14
			tween(self.countScale, 0.12, { Scale = 1 })
		end
		self:_showStack("L", self._split.L)
		self:_showStack("R", self._split.R)
		self._shownAmmo = ammo
		self.count:set(tostring(ammo))
		self.count:tint(if ammo == 0 then warnTint() else BRASS_TINT:Lerp(CHAR, math.max(0, self._toneShown)))
	end
	self._hand.L = ft.L or self._hand.L
	self._hand.R = ft.R or self._hand.R
	self.reserve:set(if ws.reserveAmmo and ws.reserveAmmo < 100000 then "/ " .. tostring(ws.reserveAmmo) else "") -- (an endless reserve: nothing)
	if ws.state == "Reloading" then
		self.status:set("RELOADING")
		self.status:tint(warnTint())
		self.status:fade(0.2 + 0.3 * (0.5 + 0.5 * math.sin(self._pulse * 1.6)))
	elseif ammo == 0 then
		self.status:set("EMPTY  -  R")
		self.status:tint(warnTint())
		self.status:fade(0)
	else
		self.status:set("")
	end
end

-- the Spin Drill's own read (3mf9): while the form lasts the sword's steel runs HOT - the guard's blade, the blade in
-- the selector and the rank letter take an ember tone with a slow breath (steady with Reduce flashing) and the blade
-- reads ASHBRAND; the guns are locked (the owner 11-01, sword only), so they burn OUT - the small crest and any rounds
-- go to charcoal and Q reads SEALED. All ease back when it ends. Orange-hot metal, never a red wash
function ChamberHud:_dtLook(dt)
	local plr = Players.LocalPlayer
	local active = self._dtOn == true
	local calm = plr:GetAttribute("S_Access_ReduceFlashing") == true
	if (self._sword and active) ~= self._nameDt then
		self._nameDt = self._sword and active
		self.weaponName:set(if self._nameDt then DT_BLADE_NAME elseif self._sword then "GREATSWORD" else "ONYX & PEARL")
	end
	if active ~= self._sealShown then
		self._sealShown = active
		self.armsKey:set(if active then "SEALED" else "Q")
		self.armsKey:tint(if active then EMBER else BONE)
	end
	self._dtTone += ((if active then 1 else 0) - self._dtTone) * math.min(1, dt * 5)
	local breath = if calm then 1 else 0.8 + 0.2 * (0.5 + 0.5 * math.sin(self._pulse * 0.55))
	local q = math.floor(self._dtTone * breath * 40 + 0.5) / 40   -- (stepped: the glyphs and rounds retint only on a change)
	-- the blade is set fresh every frame by _vitals: warm it every frame
	if q > 0 and self.guardRow.Visible then
		self.guardLit.ImageColor3 = self.guardLit.ImageColor3:Lerp(EMBER, 0.75 * q)
	end
	if q == self._toneShown then return end
	self._toneShown = q
	local white = Color3.new(1, 1, 1)
	-- (eased by the tone, not the breath: burnt-out metal does not breathe)
	local burnt = math.floor(self._dtTone * 40 + 0.5) / 40
	local roundTone = white:Lerp(CHAR, burnt)
	for _, hand in { "L", "R" } do
		for _, r in self.mags[hand].rounds do
			r.img.ImageColor3 = roundTone
		end
	end
	if self._shownAmmo ~= 0 then
		self.count:tint(BRASS_TINT:Lerp(CHAR, burnt))
	end
	local gunsBase = if self._sword then Color3.fromRGB(140, 140, 148) else white   -- (_arms' dim / lit)
	self.armsPic.guns.ImageColor3 = gunsBase:Lerp(CHAR, burnt)
	self.armsPic.sword.ImageColor3 = (if self._sword then white else Color3.fromRGB(140, 140, 148)):Lerp(EMBER, 0.6 * q)
	self.letterMain.ImageColor3 = self._letterBase:Lerp(EMBER, 0.7 * q)
	for _, l in self.letterSmall do
		l.ImageColor3 = white:Lerp(EMBER, 0.7 * q)
	end
end

function ChamberHud:update(ctl, dt)
	if not self.gui then return end
	self._pulse += dt * 7
	-- the Spin Drill: the server's DTActive on the character (the owner 11-01: sword only while it lasts), or the
	-- player's (StyleService's, round 9)
	local ch = ctl.character
	self._dtOn = (ch ~= nil and ch:GetAttribute("DTActive") == true) or Players.LocalPlayer:GetAttribute("DTActive") == true
	-- the weapon follows the slot (and the server's Weapon attribute, when it says sword; the trigger forces the sword)
	local slot = ctl.activeWeapon
	if Players.LocalPlayer:GetAttribute("Weapon") == "sword" or self._dtOn then slot = 2 end
	if slot ~= self._slot then
		self._slot = slot
		self:setWeapon(slot)
	end
	self:_vitals(ctl, dt)
	self:_style()
	self:_ammo(ctl)
	self:_dtLook(dt)
	-- the gunslinger chip: E ready, or its cooldown
	if not self._sword then
		local left = (ctl._gsReadyAt or 0) - os.clock()
		if ctl._honey then
			self.gsChip:set("HONEYCOMB FIRE")
			self.gsChip:tint(CRIMSON_TINT)
		elseif left > 0 then
			self.gsChip:set(string.format("GUNSLINGER  %.1f", left))
			self.gsChip:tint(MUTED)
		else
			self.gsChip:set("GUNSLINGER  -  E")
			self.gsChip:tint(C.Gold)
		end
	end
end

function ChamberHud:show()
	if self.gui then self.gui.Enabled = true end
end

function ChamberHud:hide()
	if self.gui then self.gui.Enabled = false end
	if self.edgeGui then self.edgeGui.Enabled = false end
end

function ChamberHud:cleanup()
	if self._matchConn then
		self._matchConn:Disconnect()
		self._matchConn = nil
	end
	if self._vpConn then
		self._vpConn:Disconnect()
		self._vpConn = nil
	end
	if self._scaleConn then
		self._scaleConn:Disconnect()
		self._scaleConn = nil
	end
	if self.gui then
		self.gui:Destroy()
		self.gui = nil
	end
	if self.edgeGui then
		self.edgeGui:Destroy()
		self.edgeGui = nil
	end
end

return ChamberHud
